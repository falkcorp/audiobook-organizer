import json,subprocess,os,re,sys
W=os.environ.get('REPO', '.')  # repo root
S=os.path.dirname(os.path.abspath(__file__))
d=json.load(open(S+'/defs.json'))
d=[r for r in d if '.test-probe-' not in r['ID']]
gofiles=subprocess.run(['git','-C',W,'ls-files','*.go'],capture_output=True,text=True).stdout.split()
gofiles=[f for f in gofiles if not f.endswith('_test.go') and '/mocks/' not in f]
src={f:open(os.path.join(W,f),errors='replace').read().split('\n') for f in gofiles}
web=subprocess.run(['git','-C',W,'ls-files','web/src'],capture_output=True,text=True).stdout.split()
web=[f for f in web if f.endswith(('.ts','.tsx')) and '.test.' not in f and '__tests__' not in f]
wsrc={f:open(os.path.join(W,f),errors='replace').read() for f in web}
# constant map: NAME = "id"
consts={}
for f,lines in src.items():
  for i,l in enumerate(lines):
    m=re.match(r"\s*(?:const\s+|var\s+)?(\w+)\s*(?:string\s*)?=\s*\"([a-z0-9]+\.[a-z0-9.\-]+)\"",l)
    if m: consts.setdefault(m.group(2),[]).append((m.group(1),f,i+1))
out={}
for r in d:
  id=r['ID']; q='"'+id+'"'
  hits=[]
  for f,lines in src.items():
    for i,l in enumerate(lines):
      if q in l: hits.append((f,i+1,l.strip()[:140]))
  names=[c[0] for c in consts.get(id,[])]
  refs=[]
  for n in names:
    pat=re.compile(r'\b'+re.escape(n)+r'\b')
    for f,lines in src.items():
      for i,l in enumerate(lines):
        if pat.search(l) and not re.match(r'\s*'+re.escape(n)+r'\s*(string\s*)?=',l): refs.append((f,i+1,l.strip()[:140]))
  # def site: hit with 'ID:' 
  defsite=[h for h in hits if re.search(r'\bID:\s',h[2])]
  if not defsite:
    defsite=[h for h in refs if re.search(r'\bID:\s',h[2])]
  if not defsite and r['RunAt'] and 'autogenerated' not in r['RunAt']:
    defsite=[(r['RunAt'].split(' ')[0].replace(W+os.sep,''),0,'run')]
  webhits=[f for f,t in wsrc.items() if id in t]
  enq=[h for h in hits+refs if 'Enqueue' in h[2] or 'enqueue' in h[2].lower() or 'Trigger' in h[2]]
  out[id]={'def':defsite[:2],'hits':hits,'refs':refs,'web':webhits,'enq':enq,'consts':names}
json.dump(out,open(S+'/loc.json','w'),indent=1)
nodef=[k for k,v in out.items() if not v['def']]
print(len(out),'nodef',len(nodef),nodef)
