import json,re,os,subprocess,collections
W=os.environ.get('REPO', '.')  # repo root
S=os.path.dirname(os.path.abspath(__file__))
defs=[r for r in json.load(open(S+'/defs.json')) if '.test-probe-' not in r['ID']]
loc=json.load(open(S+'/loc.json'))
def rd(f): return open(os.path.join(W,f),errors='replace').read()
# job id -> file
jobfile={}
for f in os.listdir(W+'/internal/maintenance/jobs'):
  if f.endswith('.go') and not f.endswith('_test.go'):
    for m in re.finditer(r'ID\(\) string\s*\{\s*return "([^"]+)"',rd('internal/maintenance/jobs/'+f)):
      jobfile['maintenance.'+m.group(1)]=('internal/maintenance/jobs/'+f, rd('internal/maintenance/jobs/'+f)[:m.start()].count('\n')+1)
# scheduler tasks
tasks=rd('internal/scheduler/tasks.go')
taskmap=collections.defaultdict(list)
for p in re.split(r'ts\.registerTask\(TaskDefinition\{',tasks)[1:]:
  nm=re.search(r'Name:\s*"([^"]+)"',p); nm=nm.group(1) if nm else None
  for e in re.findall(r'EnqueueOp\([^,]+,\s*"([\w.\-]+)"',p): taskmap[e].append(nm)
consts={'asinBackfillOpID':'metafetch.asin-backfill','candidateFetchOpID':'metadata.candidate-fetch','catalogHarvestOpID':'catalog.harvest-authors'}
for c,i in consts.items(): taskmap[i].append({'metafetch.asin-backfill':'asin_backfill','metadata.candidate-fetch':'candidate_fetch','catalog.harvest-authors':'catalog_harvest'}[i])
taskmap['dedup.rebuild-gold-labels'].append('label_refinement'); taskmap['dedup.calibrate-composite'].append('label_refinement')
# children
children={'maintenance.temp-file-cleanup':['maintenance.library-optimize'],'acoustid.fingerprint-rescan':['maintenance.library-optimize','dedup.lsh-index-build'],'acoustid.scan':['maintenance.library-optimize','dedup.run-all'],'acoustid.backfill':['maintenance.library-optimize'],'dedup.embed-scan':['dedup.run-all'],'dedup.full-scan':['dedup.run-all'],'dedup.llm-review':['dedup.run-all'],'dedup.rescore':['dedup.rescore(self-chain)'] ,'metadata.candidate-fetch':['repairs fixers (author-named-series, scan-title-revert)'],'library.bulk-metadata-fetch':['library.bulk-metadata-fetch (self-chain)']}
uimap={}
gofiles=subprocess.run(['git','-C',W,'ls-files','*.go'],capture_output=True,text=True).stdout.split()
gofiles=[f for f in gofiles if not f.endswith('_test.go') and '/mocks/' not in f]
rows=[]
for r in defs:
  id=r['ID']; L=loc[id]
  if id in jobfile: deff,defl=jobfile[id]
  elif L['def'] and L['def'][0][1]: deff,defl=L['def'][0][0],L['def'][0][1]
  else: deff,defl=(L['def'][0][0],0) if L['def'] else ('?',0)
  # run file
  ra=r['RunAt'].split(' ')[0].replace(W+os.sep,'')
  runf=deff if (not ra or 'autogenerated' in ra or 'maintenance_job_op' in ra) else ra.split(':')[0]
  # RunItems in run file
  try: txt=rd(runf)
  except: txt=''
  ri=len(re.findall(r'RunItems\(',txt))
  conc=sorted(set(m.strip() for m in re.findall(r'Concurrency:\s*([^,\n]+)',txt)))
  trig=[]
  if id in taskmap: trig.append('sched-task:'+'/'.join(taskmap[id]))
  if r['Schedule']: trig.append('cron-decl(unread):'+r['Schedule'])
  if r['Triggers']: trig.append('event:'+','.join(r['Triggers']))
  if id in children: trig.append('child-of:'+'/'.join(children[id]))
  if id in jobfile: trig.append('UI:/maintenance/jobs')
  # handler/web
  extern=[h for h in L['hits']+L['refs'] if 'Enqueue' in h[2] and not h[0].startswith('internal/scheduler/')]
  hdl=sorted(set(h[0].split('/')[-1]+':'+str(h[1]) for h in extern))
  if hdl: trig.append('code-enqueue:'+','.join(hdl[:3]))
  if L['web']: trig.append('web:'+','.join(sorted(set(os.path.basename(w) for w in L['web']))[:3]))
  if not trig: trig.append('API-only(POST /operations/v2)')
  rows.append(dict(id=id,plugin=r['Plugin'] or '(server)',loc=f'{deff}:{defl}',name=r['DisplayName'],desc=(r['Description'] or '').split('. ')[0][:150],
    trig='; '.join(trig),ri=ri,conc=','.join(conc)[:60],resume=r['ResumePolicy'],cancel=r['Cancellable'],live=r['Liveness'],key=r['ConcurrencyKey'],sched=r['Schedule'],writes=','.join(r['Writes'] or []),timeout=r['Timeout']))
json.dump(rows,open(S+'/rows.json','w'),indent=1)
c=collections.Counter('API-only' in x['trig'] for x in rows); print(c)
print(sum(1 for x in rows if x['ri']>0),'files w RunItems')
